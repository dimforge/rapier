# Compiles and runs the C snippets of the user-guide against the in-tree C bindings.
cmake_minimum_required(VERSION 3.20)
project(RapierCSnippets2D LANGUAGES C CXX)
set(RAPIER_DIMENSION "2" CACHE STRING "" FORCE)
set(RAPIER_FEATURES "fem" CACHE STRING "" FORCE)
set(RAPIER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(RAPIER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../../c" rapier)
enable_testing()
file(GLOB SNIPPETS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/examples/*.c")
foreach(snippet ${SNIPPETS})
  get_filename_component(name "${snippet}" NAME_WE)
  rapier_executable(${name} "${snippet}")
  add_test(NAME ${name} COMMAND ${name} WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endforeach()
