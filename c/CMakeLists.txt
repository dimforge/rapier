  message(FATAL_ERROR "8-lane SIMD requires RAPIER_PRECISION=32; f64 uses 4 lanes")

  message(FATAL_ERROR "8-lane SIMD is incompatible with enhanced-determinism; use RAPIER_SIMD_LANES=4")

function(rapier_executable name source)
  add_executable(${name} ${source})
  target_link_libraries(${name} PRIVATE Rapier::rapier)
  target_compile_features(${name} PRIVATE c_std_11 cxx_std_17)
  if(MSVC)
    target_compile_options(${name} PRIVATE /W4 /WX /UNDEBUG)
  else()
    target_compile_options(${name} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  endif()
  if(WIN32 AND RAPIER_SHARED)
    add_custom_command(TARGET ${name} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${RAPIER_LIBRARY}" "$<TARGET_FILE_DIR:${name}>" VERBATIM)
  endif()
endfunction()
